```bash
#!/usr/bin/env bash
# Open an AWS SSM Session Manager interactive shell to the operator's devbox.
# Replaces the old `ssh -i ~/.ssh/${KEY_NAME}.pem ec2-user@${PUBLIC_IP}` flow now
# that Phase 2 closed :22 ingress on the security group.
#
# See: .planning/phases/02-network-exposure-remediation/02-RESEARCH.md (Pattern 5)
# Requires: session-manager-plugin (https://docs.aws.amazon.com/systems-manager/latest/userguide/session-manager-working-with-install-plugin.html)

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
source "$SCRIPT_DIR/_common.sh"

usage() {
  cat <<EOF
Usage: $(basename "$0") [OPTIONS]

Open an SSM Session Manager shell to the operator's devbox.

Options:
  --user USERNAME     Devbox owner (default: \$DEVBOX_USER or \$(whoami))
  --instance-id ID    EC2 instance ID (overrides Terraform state)
  --region REGION     AWS region (overrides Terraform state)
  -h, --help          Show this help

Requirements (operator workstation):
  aws CLI >= 1.16.12 (you already have this if `make build` works)
  session-manager-plugin (brew install --cask session-manager-plugin on macOS)
EOF
  exit 0
}

parse_args "$@"

# Pre-flight: session-manager-plugin must be installed on the operator workstation.
# The plugin is separate from the AWS CLI itself; aws ssm start-session shells out to it.
if ! command -v session-manager-plugin >/dev/null 2>&1; then
  cat >&2 <<EOF
ERROR: session-manager-plugin is not installed on this workstation.

Install (macOS):
  brew install --cask session-manager-plugin

Install (Linux / other):
  https://docs.aws.amazon.com/systems-manager/latest/userguide/session-manager-working-with-install-plugin.html

Verify after install:
  session-manager-plugin
  # Expected: "The Session Manager plugin was installed successfully..."
EOF
  exit 1
fi

init_devbox

echo "Opening SSM session to ${INSTANCE_ID} in ${REGION}..."
echo "(Type 'exit' or Ctrl-D to end the session.)"
echo ""

exec aws ssm start-session \
  --target "$INSTANCE_ID" \
  --region "$REGION"
```

Implementation notes:
- `init_devbox` (from `_common.sh`) sets `INSTANCE_ID` and `REGION` via `terragrunt output -raw instance_id` / `terragrunt output -raw aws_region`. Both outputs exist in 02-01's Terraform layer.
- The pre-flight `command -v session-manager-plugin` check runs BEFORE `init_devbox` so an operator on a fresh workstation gets the actionable error before the script reaches into tfstate (which might itself fail with a confusing message). The `exit 1` keeps the failure mode tight.
- `exec aws ssm start-session ...` — execs so the child becomes the shell process; the operator's Ctrl-C / Ctrl-D goes directly to the session, not the wrapper.
- No `--document-name` is specified — uses the default `SSM-SessionManagerRunShell` document, which lands as `ssm-user` (RESEARCH.md:565-568). Operators who want to land as `ec2-user` can `sudo -iu ec2-user` once in the shell; Phase 4 CLAUDE.md will document this. We deliberately do NOT swap to `AWS-StartSSHSession` here because that requires the operator to have `~/.ssh/config` set up with a `ProxyCommand` block — that's an advanced path, not the daily UX.

File content:

```bash
#!/usr/bin/env bash
# Resolve the operator's current public IP and write a gitignored
# allowlist.auto.tfvars that populates var.allowed_web_cidrs with that /32.
#
# Phase 2 hybrid posture: code-server (:8080) and noVNC (:6080) ingress is
# restricted to var.allowed_web_cidrs (terraform/variables.tf, added in 02-01).
# This script is the operator's one-step refresh path when their public IP
# changes (coffee shop, hotel WiFi, mobile tether).
#
# Output:  ./allowlist.auto.tfvars (gitignored — see .gitignore Phase-2 stanza)
# Next:    make tg-apply

set -euo pipefail

usage() {
  cat <<EOF
Usage: $(basename "$0") [OPTIONS]

Resolve your current public IP from https://checkip.amazonaws.com and write a
gitignored allowlist.auto.tfvars populating var.allowed_web_cidrs with that /32.

Options:
  --output PATH       Output path (default: ./allowlist.auto.tfvars)
  --extra-cidr CIDR   Additional CIDR to include (repeatable — e.g. office range)
  -h, --help          Show this help

Behind a captive portal? Override manually:
  echo 'allowed_web_cidrs = ["YOUR.IP.HERE/32"]' > allowlist.auto.tfvars
EOF
  exit 0
}

TFVARS_PATH="${TFVARS_PATH:-./allowlist.auto.tfvars}"
EXTRA_CIDRS=()

while [[ $# -gt 0 ]]; do
  case $1 in
    --output)      TFVARS_PATH="$2"; shift 2 ;;
    --extra-cidr)  EXTRA_CIDRS+=("$2"); shift 2 ;;
    -h|--help)     usage ;;
    *)             echo "Unknown option: $1" >&2; exit 1 ;;
  esac
done

# Resolve operator's current public IP via AWS-hosted echo service.
# --max-time 5 bounds failures (captive portal hangs, DNS issues).
echo "Resolving public IP from https://checkip.amazonaws.com ..."
PUBLIC_IP=$(curl -sS --max-time 5 https://checkip.amazonaws.com | tr -d '[:space:]')

# Defense against captive-portal HTML: the response must look exactly like an
# IPv4 address. The Terraform `cidrhost()` validation in
# terraform/variables.tf is the second layer; this is the first.
if [[ ! "$PUBLIC_IP" =~ ^([0-9]{1,3}\.){3}[0-9]{1,3}$ ]]; then
  cat >&2 <<EOF
ERROR: checkip.amazonaws.com returned unexpected payload: '$PUBLIC_IP'

Are you behind a captive portal, or is curl blocked?
Override manually:
  echo 'allowed_web_cidrs = ["YOUR.IP.HERE/32"]' > $TFVARS_PATH
EOF
  exit 1
fi

# Build the CIDR list — operator's /32 plus any --extra-cidr the operator passed.
CIDR_LIST="\"${PUBLIC_IP}/32\""
for c in "${EXTRA_CIDRS[@]}"; do
  CIDR_LIST="${CIDR_LIST}, \"${c}\""
done

# Atomic write — write to a temp file, then mv into place.
TMP_FILE="${TFVARS_PATH}.tmp.$$"
cat > "$TMP_FILE" <<EOF
# AUTO-GENERATED by make devbox-allowlist-me on $(date -u +%Y-%m-%dT%H:%M:%SZ)
# Operator: ${USER:-unknown}
# Public IP at time of write: ${PUBLIC_IP}
# Refresh by re-running 'make devbox-allowlist-me' after IP changes.
# Gitignored — see .gitignore Phase-2 stanza.
allowed_web_cidrs = [${CIDR_LIST}]
EOF
mv "$TMP_FILE" "$TFVARS_PATH"

echo ""
echo "Wrote ${TFVARS_PATH}:"
echo "----------------------------------------"
cat "$TFVARS_PATH"
echo "----------------------------------------"
echo ""
echo "Next: make tg-apply"
```

Implementation notes:
- The script does NOT source `_common.sh` because it doesn't need to resolve an instance — it operates purely on the operator workstation. Keep it standalone.
- Atomic write (`cat > $TMP; mv $TMP $TFVARS_PATH`) prevents a partial-write race if the operator Ctrl-Cs mid-execution.
- Output path defaults to `./allowlist.auto.tfvars` (repo root), which is where Terragrunt loads `*.auto.tfvars` from for the root terragrunt.hcl (RESEARCH.md A5 line 560 flags this as needing verification; we verify in Task 4 by running an apply against a test value — see plan verification section).
- `--extra-cidr` flag (repeatable) is for operators who want to add an office IP range or a colleague's IP without overwriting their own — power-user feature; default flow ignores it.
- The IPv4 regex is intentionally loose (doesn't validate 0-255 per octet) because the Terraform `cidrhost()` validation in variables.tf catches any garbage that gets through. Defense in depth, not defense in single-grep.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| operator workstation → external internet | `scripts/devbox-allowlist-me.sh` makes an HTTPS GET to `checkip.amazonaws.com`. Untrusted response payload crosses into the script. |
| operator workstation → AWS SSM service | `scripts/devbox-ssm.sh` and the `devbox-port-forward` recipe shell out to `aws ssm start-session`, which opens a TLS WebSocket to the AWS control plane. Operator IAM principal authenticates via SigV4. |
| operator workstation → operator filesystem | `scripts/devbox-allowlist-me.sh` writes `./allowlist.auto.tfvars` containing the resolved /32. |
| Makefile recipe → terragrunt CLI | `devbox-port-forward` recipe reads `terragrunt output -raw instance_id` / `aws_region` from tfstate. tfstate is the source of truth; recipe doesn't validate it. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-10 | Tampering | `scripts/devbox-allowlist-me.sh` reading captive-portal HTML and writing it into tfvars | mitigate | IPv4-shape regex (`^([0-9]{1,3}\.){3}[0-9]{1,3}$`) runs BEFORE the write. `curl --max-time 5` bounds the request. Atomic temp-file write (`mv $TMP $TFVARS_PATH`) prevents partial-write on Ctrl-C. Defense in depth: Terraform `cidrhost()` validation in `terraform/variables.tf` (02-01 Task 1) is the second layer — even a maliciously crafted string that beats the bash regex still hits the Terraform validation block. |
| T-02-11 | Information Disclosure | Operator's public IP (or office IP range with `--extra-cidr`) leaking into git | mitigate | `.gitignore` entry `*.auto.tfvars` and `allowlist.auto.tfvars` (added by 02-01 Task 4). The atomic write writes ONLY to `./allowlist.auto.tfvars` (not a randomly-suggested filename), so the gitignore is structurally correct. |
| T-02-12 | Spoofing | `checkip.amazonaws.com` DNS poisoning / man-in-the-middle returning attacker-controlled IP | accept | HTTPS verification (curl's default cert validation) protects against MITM. DNS poisoning would still require the attacker's IP to (a) pass the IPv4 regex, (b) be one the attacker controls, and (c) be a value the operator wants in their allowlist — at which point the threat model is "the operator's DNS is owned," which has worse blast radii than a wrong allowlist entry. Acceptable for a personal devbox. |
| T-02-13 | Denial of Service | `make devbox-ssm` invoked when session-manager-plugin missing | mitigate | Pre-flight `command -v session-manager-plugin` check prints brew/Linux install recipe to stderr and exits 1. Same pattern in `devbox-port-forward` recipe. No silent failure mode. |
| T-02-14 | Elevation of Privilege | Operator runs `make devbox-port-forward` while attacker has access to `localhost:8080` | mitigate | code-server / noVNC require their Phase 1 random passwords (SSM Parameter Store SecureStrings). The port forward exposes the same authentication surface as direct ingress would — no privilege escalation introduced. Recommend in 02-SUMMARY.md and Phase 4 CLAUDE.md that operators close port-forward sessions when not in use (Ctrl-C in the terminal). |
| T-02-15 | Repudiation | Operator denies starting a port-forward session | mitigate | `aws ssm start-session` with `--document-name AWS-StartPortForwardingSession` IS logged to CloudTrail (same path as a shell session). Document type is visible in the event, distinguishing shell from port-forward. Audit story is identical to T-02-08 in 02-01. |
| T-02-16 | Information Disclosure | `make devbox-port-forward` exposes :8080 on the operator's `0.0.0.0:8080` to anyone on the operator's LAN | accept | `aws ssm start-session ... AWS-StartPortForwardingSession` by default binds to `127.0.0.1` (loopback), not `0.0.0.0`. Verified in RESEARCH.md:528-533 and AWS docs. Operators who explicitly want a LAN-accessible forward must opt in via `localBindAddress` parameter; the recipe in this plan does not pass `localBindAddress`, so the loopback-only default applies. Acceptable. |
